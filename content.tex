\eintrag{ADC}[
  lang  = Analog-to-Digital Converter,
  dt    = Analog-Digital-Umsetzer (ADU),
  auch  = A2D,
]{Wandelt ein analoges Messsignal in ein digitales Signal um. Dazu wird das Signal abgetastet (zeitliche Diskretisierung) und quantisiert (Diskretisierung der Amplitude).}

\eintrag{ARM}[
  lang  = Advanced RISC Machine,
]{Prozessorarchitektur nach dem RISC-Prinzip (Reduced Instruction Set Computer): wenige, einfache Befehle. Weit verbreitet in Mobilgeräten und eingebetteten Systemen.}

\eintrag{ASCII}[
  lang  = American Standard Code for Information Interchange,
]{7-Bit-Zeichencodierung mit 128 Zeichen: lateinische Buchstaben, Ziffern, Satz- und Steuerzeichen. Enthält keine Umlaute.}

\eintrag{ASIC}[
  lang  = Application-Specific Integrated Circuit,
]{Ist ein IC, der speziell für eine klar definierte Aufgabe entwickelt wurde. Er ist nicht frei programmierbar.}

\eintrag{ASP}[
  lang  = Average Selling Price,
  dt    = durchschnittlicher Verkaufspreis,
]{Mittlerer Preis, zu dem ein Produkt über alle Kunden und Stückzahlen hinweg verkauft wird. Bestimmt zusammen mit Stückzahl und NRE-Kosten die Wirtschaftlichkeit eines Produkts.}

\eintrag{ASSP}[
  lang  = Application-Specific Standard Product,
]{Ist ein IC, der für einen bestimmten Anwendungsbereich entwickelt wurde, im Gegensatz zu einem ASIC aber frei verkäuflich ist.}

\eintrag{BCD}[
  lang  = Binary Coded Decimal,
  dt    = binär codierte Dezimalzahl,
]{Jede Dezimalziffer wird einzeln als 4-Bit-Binärzahl (Tetrade) dargestellt.}

\eintrag{Bus}
    {Gemeinsamer Übertragungsweg, über den mehrere Komponenten Daten austauschen.}

\eintrag{C2C}[
  lang  = Car-to-Car,
  dt    = Fahrzeug-zu-Fahrzeug-Kommunikation,
  auch  = V2V (Vehicle-to-Vehicle),
]{Fahrzeuge tauschen direkt Daten aus, etwa Warnungen vor Gefahrenstellen.}

\eintrag{C2I}[
  lang  = Car-to-Infrastructure,
  dt    = Fahrzeug-zu-Infrastruktur-Kommunikation,
  auch  = V2I (Vehicle-to-Infrastructure),
]{Fahrzeuge tauschen Daten mit der Verkehrsinfrastruktur aus, etwa mit Ampeln oder Verkehrsleitsystemen.}

\eintrag{Cache}
    {Schneller Zwischenspeicher für konfliktfreien Zugriff auf häufig benötigte Daten. Funktion: Entlastung des langsameren Hauptspeichers, somit Reduzierung von Ladezeiten und Verbesserung der Systemperformanz.}

\eintrag{CCD}[
  lang  = Charge-Coupled Device,
  dt    = ladungsgekoppeltes Bauelement,
]{Bildsensor, dessen Pixel einfallendes Licht in elektrische Ladung umwandeln. Die Ladungen werden schrittweise zum Ausgang verschoben und dort ausgelesen.}

\eintrag{CMOS}[
  lang  = Complementary Metal-Oxide-Semiconductor,
  dt    = komplementärer Metall-Oxid-Halbleiter,
]{Bauweise für Logikschaltungen, bei der immer zwei gegensätzlich schaltende Transistoren ein Paar bilden: Ist der eine an, ist der andere aus. Dadurch fließt fast nur im Moment des Umschaltens Strom. Standardtechnik heutiger ICs.}

\eintrag{CPU}[
  lang  = Central Processing Unit,
  auch  = Prozessor,
]{Hardwarekomponente, führt mathematische und logische Operationen durch. Besteht aus ALU, Steuerwerk und Registern.}

\eintrag{DAC}[
  lang  = Digital-to-Analog Converter,
  dt    = Digital-Analog-Umsetzer (DAU),
  auch  = D2A,
]{Wandelt ein digitales Signal in ein analoges Signal (Spannung oder Strom) um. Gegenstück zum ADC.}

\eintrag{DB}[
  lang  = Datenbank,
]{Strukturierte, dauerhaft gespeicherte Sammlung von Daten. Wird von einem Datenbankmanagementsystem (DBMS) verwaltet, das Abfragen, Änderungen und gleichzeitigen Zugriff mehrerer Nutzer regelt.}

\eintrag{DMA}[
  lang  = Direct Memory Access,
  dt    = Speicherdirektzugriff,
]{Peripheriegeräte tauschen Daten direkt mit dem Hauptspeicher aus, ohne dass die CPU jedes Datenwort selbst übertragen muss. Entlastet so die CPU.}

\eintrag{DRAM}[
  lang  = Dynamic Random Access Memory,
  dt    = dynamischer RAM,
]{Speichert jedes Bit als Ladung in einem winzigen Kondensator. Die Ladung entweicht langsam, deshalb muss der Inhalt ständig aufgefrischt werden. Günstig und platzsparend, daher als Arbeitsspeicher im PC genutzt.}

\eintrag{EAC}[
  lang  = End-Around-Carry,
  dt    = zyklischer Übertrag,
]{Rechenregel beim Einerkomplement, in dem negative Zahlen durch Umdrehen aller Bits gebildet werden: Entsteht beim Addieren ein Übertrag über die höchste Stelle hinaus, wird er nicht verworfen, sondern zur niedrigsten Stelle hinzuaddiert.}

\eintrag{EEPROM}[
  lang  = Electrically Erasable Programmable Read-Only Memory,
  dt    = elektrisch löschbarer programmierbarer Nur-Lese-Speicher,
]{Nichtflüchtiger Speicher, der im Gegensatz zum EPROM elektrisch gelöscht wird, byteweise und ohne Ausbau aus der Schaltung.}

\eintrag{ENIAC}[
  lang  = Electronic Numerical Integrator and Computer,
]{Gilt als erster elektronischer Universalrechner (1946, USA). Arbeitete mit Elektronenröhren und wurde durch Umstecken von Kabeln programmiert.}

\eintrag{EPROM}[
  lang  = Erasable Programmable Read-Only Memory,
  dt    = löschbarer programmierbarer Nur-Lese-Speicher,
]{Nichtflüchtiger Speicher, der elektrisch beschrieben und mit UV-Licht durch ein Quarzfenster im Gehäuse gelöscht wird.}

\eintrag{Feldbus}
    {Bus in der Automatisierungstechnik, der Feldgeräte (Sensoren, Aktoren) mit einer Steuerung verbindet. Ersetzt die Einzelverdrahtung jedes Geräts.}

\eintrag{FPGA}[
  lang  = Field-Programmable Gate Array,
]{Integrierter Schaltkreis, dessen Logikfunktion erst nach der Fertigung durch eine Konfigurationsdatei festgelegt wird.}

\eintrag{FSM}[
  lang  = Finite State Machine,
  dt    = endlicher Automat,
]{Modell eines Systems mit endlich vielen Zuständen. Abhängig vom aktuellen Zustand und der Eingabe wechselt es in einen Folgezustand und erzeugt eine Ausgabe.}

\eintrag{H2H}[
  lang  = Human-to-Human,
  dt    = Mensch-zu-Mensch-Kommunikation,
]{Menschen kommunizieren über ein Netzwerk miteinander.}

\eintrag{H2T}[
  lang  = Human-to-Thing,
  dt    = Mensch-zu-Ding-Kommunikation,
]{Ein Mensch steuert ein vernetztes Gerät oder ruft dessen Daten ab.}

\eintrag{HW/SW-Cosimulation}[
  lang  = Hardware/Software-Cosimulation,
]{Gemeinsame Simulation von Hardware und Software eines Systems, bevor die Hardware physisch existiert. Ziel: Fehler im Zusammenspiel früh finden und HW- und SW-Entwicklung parallelisieren.}

\eintrag{IC}[
  lang  = Integrated Circuit,
]{Eine elektronische Schaltung, die auf einem kleinen Plättchen aus Halbleitermaterial (meist Silizium, engl. Silicon) aufgebracht ist.}

\eintrag{IEEE}[
  lang  = Institute of Electrical and Electronics Engineers,
]{Weltweiter Berufsverband für Elektrotechnik und Informatik, der u.\,a. technische Standards herausgibt, etwa IEEE 754 für Gleitkommazahlen.}

\eintrag{IEEE 754}[
  lang  = IEEE Standard for Floating-Point Arithmetic,
]{Norm für die binäre Darstellung von Gleitkommazahlen aus Vorzeichen, Exponent und Mantisse. Gängige Formate: 32 Bit (Single Precision) und 64 Bit (Double Precision).}

\eintrag{I/O}[
  lang  = Input/Output,
  dt    = Eingabe/Ausgabe,
]{Datenaustausch.}

\eintrag{IoT}[
  lang  = Internet of Things,
  dt    = Internet der Dinge,
]{Vernetzung physischer Geräte mit Sensoren und Aktoren, sodass sie über das Internet Daten austauschen und selbstständig zusammenarbeiten. Beispiel: Smart Home, etwa ein Heizungsthermostat, das sich per App steuern lässt und automatisch herunterregelt, wenn ein Fensterkontakt ein offenes Fenster meldet.}

\eintrag{ISA}[
  lang  = Industry Standard Architecture,
  id    = ISA-Bus,
]{Bus-Standard für Erweiterungskarten im IBM-PC (ab 1981). Später durch PCI abgelöst.}

\eintrag{ISA}[
  lang  = Instruction Set Architecture,
  dt    = Befehlssatzarchitektur,
]{Legt fest, welche Befehle, Datentypen und Adressierungsarten ein Prozessor versteht. Beispiele: x86, x86-64, ARM.}

\eintrag{JPEG}[
  lang  = Joint Photographic Experts Group,
]{Gremium und das nach ihm benannte, verlustbehaftete Kompressionsverfahren für Fotos. Spart Speicher, indem feine Bilddetails verworfen werden, die das Auge kaum wahrnimmt.}

\eintrag{Kritischer Pfad}
    {Langsamster Signalweg durch eine digitale Schaltung. Er legt fest, wie schnell die Schaltung höchstens getaktet werden darf, denn das Signal muss diesen Weg innerhalb eines Taktes durchlaufen.}

\eintrag{LCD}[
  lang  = Liquid Crystal Display,
  dt    = Flüssigkristallanzeige,
]{Flüssigkristalle ändern unter Spannung ihre Ausrichtung und lassen so das Licht einer Hintergrundbeleuchtung durch oder sperren es. Leuchtet nicht selbst.}

\eintrag{LED}[
  lang  = Light-Emitting Diode,
  dt    = Leuchtdiode,
  auch  = Lichtemitterdiode,
]{Halbleiterbauelement, das Licht abgibt, wenn Strom in Durchlassrichtung fließt. Die Farbe hängt vom Halbleitermaterial ab.}

\eintrag{LSB}[
  lang  = Least Significant Bit,
  dt    = niederwertigstes Bit,
]{Bit mit der kleinsten Stellenwertigkeit ($2^0$), in der üblichen Schreibweise ganz rechts.}

\eintrag{MCU}[
  lang  = Microcontroller Unit,
  dt    = Mikrocontroller,
]{Vereint CPU, Speicher und Peripherie auf einem einzigen IC. Typischer Rechenkern eingebetteter Systeme.}

\eintrag{MIPS}[
  lang  = Million Instructions Per Second,
]{Maßeinheit für die Rechenleistung eines Prozessors.}

\eintrag{MOSFET}[
  lang  = Metal-Oxide-Semiconductor Field-Effect Transistor,
  dt    = Metall-Oxid-Halbleiter-Feldeffekttransistor,
]{Häufigste Transistorart. Eine Spannung am Steueranschluss (Gate) schaltet den Strom zwischen den beiden anderen Anschlüssen ein oder aus. Da das Gate durch eine dünne Isolierschicht abgetrennt ist, braucht das Schalten selbst fast keinen Strom. Grundbaustein von CMOS.}

\eintrag{MSB}[
  lang  = Most Significant Bit,
  dt    = höchstwertiges Bit,
]{Bit mit der größten Stellenwertigkeit, in der üblichen Schreibweise ganz links. Zeigt bei vorzeichenbehafteten Zahlen das Vorzeichen an.}

\eintrag{NRE}[
  lang  = Non-Recurring Engineering Costs,
]{Einmalige Kosten für Entwicklung, Verifikation und Fertigungsvorbereitung, die unabhängig von der Stückzahl anfallen.}

\eintrag{Osc}[
  lang  = Oscillator,
  dt    = Oszillator,
]{Schaltung, die ein periodisches Signal erzeugt. Dient in digitalen Systemen als Taktgeber.}

\eintrag{OV/OPV}[
  lang  = Operationsverstärker,
  auch  = Op-Amp (Operational Amplifier),
]{Verstärkerbaustein mit zwei Eingängen, der den Spannungsunterschied zwischen ihnen sehr stark verstärkt. Je nachdem, wie man ihn mit Widerständen und Kondensatoren verschaltet, arbeitet er als Verstärker, Vergleicher oder Filter.}

\eintrag{PCI}[
  lang  = Peripheral Component Interconnect,
]{Paralleler Bus-Standard für Erweiterungskarten im PC (ab 1992). Nachfolger von ISA, später durch das serielle PCI Express (PCIe) abgelöst.}

\eintrag{PLD}[
  lang  = Programmable Logic Device,
  dt    = programmierbarer Logikbaustein,
]{IC, dessen Logikfunktion nicht schon bei der Fertigung feststeht, sondern vom Anwender festgelegt wird. Oberbegriff, zu dem auch das FPGA gehört.}

\eintrag{Polling}[
  dt    = zyklische Abfrage,
]{Die CPU fragt ein Gerät in festen Abständen immer wieder, ob neue Daten vorliegen. Einfach umzusetzen, kostet aber Rechenzeit, auch wenn nichts anliegt. Gegenstück: Interrupt, bei dem sich das Gerät von selbst meldet.}

\eintrag{PROM}[
  lang  = Programmable Read-Only Memory,
  dt    = programmierbarer Nur-Lese-Speicher,
]{Kann nach der Fertigung genau einmal beschrieben werden und ist danach unveränderlich. Vorstufe von EPROM und EEPROM, die sich wieder löschen lassen.}

\eintrag{RAM}[
  lang  = Random Access Memory,
  dt    = Speicher mit wahlfreiem Zugriff,
]{Jede Speicherzelle ist direkt über ihre Adresse lesbar und beschreibbar. Flüchtig, d.\,h. der Inhalt geht ohne Stromversorgung verloren. Dient als Arbeitsspeicher.}

\eintrag{Rapid Prototyping}[
  dt    = schneller Prototypenbau,
]{Sammelbegriff für Verfahren, mit denen schnell und kostengünstig physische Prototypen sowie Test- und Prüfkörper aus digitalen 3D-CAD-Modellen erstellt werden.}

\eintrag{RMS}[
  lang  = Root Mean Square,
  dt    = Effektivwert,
]{Gibt an, wie stark eine Wechselspannung im Mittel wirkt. Beispiel: 230\,V Wechselspannung aus der Steckdose erwärmen eine Herdplatte genauso stark wie konstante 230\,V Gleichspannung.}

\eintrag{SI}[
  lang  = Système international d’unités,
  dt    = Internationales Einheitensystem,
]{Weltweit gültiges System physikalischer Einheiten, aufgebaut auf sieben Basiseinheiten: Meter, Kilogramm, Sekunde, Ampere, Kelvin, Mol und Candela.}

\eintrag{SRAM}[
  lang  = Static Random Access Memory,
  dt    = statischer RAM,
]{Speichert jedes Bit in einer kleinen Schaltung aus mehreren Transistoren. Behält die Daten, solange Strom anliegt, ohne aufgefrischt zu werden. Schnell, aber teuer und platzintensiv, daher vor allem als Cache genutzt.}

\eintrag{T2T}[
  lang  = Thing-to-Thing,
  dt    = Ding-zu-Ding-Kommunikation,
  auch  = M2M (Machine-to-Machine),
]{Vernetzte Geräte tauschen Daten ohne menschliches Zutun aus. Grundlage des IoT.}

\eintrag{Tiefpass}[
  auch  = Tiefpassfilter,
]{Ein elektronischer Filter, der Signalanteile unterhalb seiner Grenzfrequenz passieren lässt und solche oberhalb dieser blockiert.}

\eintrag{Transistor}
    {Halbleiterbauelement, das als elektronischer Schalter oder Verstärker arbeitet: Ein kleines Steuersignal regelt einen größeren Strom. Grundbaustein aller digitalen Schaltungen.}

\eintrag{TTL}[
  lang  = Transistor-Transistor-Logik,
]{Bauweise für Logikschaltungen aus Bipolartransistoren, die anders als der MOSFET über einen Steuerstrom geschaltet werden. Arbeitet mit 5\,V. Verbraucht auch im Ruhezustand Strom und wurde daher weitgehend durch CMOS verdrängt.}

\eintrag{TTM}[
  lang  = Time-to-Market,
]{Zeitspanne von der Produktidee bis zur Markteinführung.}

\eintrag{UART}[
  lang  = Universal Asynchronous Receiver-Transmitter,
]{Baustein für serielle Datenübertragung. Asynchron, d.\,h. ohne gemeinsame Taktleitung: Sender und Empfänger vereinbaren vorab die Baudrate, Start- und Stoppbits rahmen jedes Datenwort ein.}
